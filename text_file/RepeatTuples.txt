# Creating two tuples for summer and winter clothes
summer_clothes = ("T-Shirt", "Shorts", "Sandals")
winter_clothes = ("Jacket", "Sweater", "Boots")

# Concatenating tuples to create a full inventory list
full_inventory = summer_clothes + winter_clothes
print("Full Clothing Inventory:", full_inventory)

# Repeating a specific clothing item (e.g., customer buys 5 T-Shirts)
tshirt_order = ("T-Shirt",) * 5
print("Customer's T-Shirt Order:", tshirt_order)
